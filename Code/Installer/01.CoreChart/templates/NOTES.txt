-----------------------------------------------------------------------

1. Check installation status:
   kubectl get pods -n {{ .Release.Namespace }} -l "app.kubernetes.io/instance={{ .Release.Name }}"

2. Access Argo CD:
   {{- if (index .Values "argo-cd").enabled }}
   Get the Argo CD NodePort:
   export ARGO_PORT=$(kubectl get --namespace {{ (index .Values "argo-cd").namespaceOverride | default "argocd" }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "core-umbrella-chart.fullname" . }}-argo-cd-server)
   echo http://$NODE_IP:$ARGO_PORT

   Get the initial admin password:
   kubectl -n {{ (index .Values "argo-cd").namespaceOverride | default "argocd" }} get secret argocd-initial-admin-secret -o jsonpath="{.data.password}" | base64 -d; echo
   {{- end }}
